# Edit Page Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the set list editor with a Simple pattern and a Complex cycle list that share one tempo and expand to the same playback segments.

**Architecture:** `Session` stores shared tempo fields plus a Simple pattern and a cycle list. `expandSession` resolves whichever `type` is selected into count-in, practice, and rest segments. `loadSession` migrates an old `sets` document into Complex once. Edit writes that session; Practice plays it and commits in-place edits back onto the same fields.

**Tech Stack:** Vite 6, React 19, TypeScript, Vitest 3, Testing Library, user-event.

## Global Constraints

- One session BPM. Played tempo is `min(300, session BPM + BPM increase × floor(practice index / every))`, index starting at 0.
- Simple hides Every. Every is the Practices count. Complex shows Every, an integer of at least 1.
- A length is clock time or bars. A bar uses the session meter. Rests are clock time. A time length does not change when the tempo changes.
- The last practice has no rest. A rest of 0 is omitted, and the next count-in still plays. Count-in stays four beats in 4/4.
- Simple to Complex rebuilds cycles from the pattern and keeps the tempos. A visit that opened on Complex keeps those cycles when Simple is only previewed.
- Done saves the type on screen. The stage, transport, and strip stay where they are. No look-ahead scheduling.
- BPM is an integer from 40 to 300. The played tempo caps at 300.
- Heading accessible name stays Mentronome. Empty name displays Practice.
- Editor column max width stays 960px. Below 800px the two cards stack. At 800px and above they sit side by side at equal height.

```ts
import { CLICKS, COLORS, METERS, type ClickId, type Cycle, type Length, type Meter, type Session, type SetColor, type SimplePattern } from "./types"

const integer = (value: number) => Math.round(Number.isFinite(value) ? value : 0)

export function sessionTitle(session: Session): string {
  return session.name.trim() || "Practice"
}

export function clampBpm(bpm: number): number {
  return Math.min(300, Math.max(40, integer(bpm)))
}

export function freshSimple(): SimplePattern {
  return {
    practices: 1,
    length: { unit: "time", seconds: 120 },
    restAfterSec: 30,
    stepTempo: false,
    times: 2,
    repeatRound: false,
    rounds: 2,
    restBetweenSec: 30,
  }
}

export function createDefaultSession(nextId: () => string = () => crypto.randomUUID()): Session {
  return {
    name: "Practice",
    color: "red",
    meter: "4/4",
    bpm: 80,
    bpmIncrease: 0,
    every: 1,
    click: "click-1",
    type: "simple",
    simple: freshSimple(),
    cycles: [{ id: nextId(), length: { unit: "time", seconds: 120 }, restSec: 30 }],
  }
}

export function ladderBpm(bpm: number, increase: number, every: number, index: number): number {
  const step = Math.max(1, integer(every))
  return Math.min(300, clampBpm(bpm) + Math.max(0, integer(increase)) * Math.floor(index / step))
}

export function beatsInMeter(meter: Meter): number {
  return Number(meter.split("/")[0])
}

export function lengthSeconds(length: Length, meter: Meter, bpm: number): number {
  if (length.unit === "time") return Math.max(1, integer(length.seconds))
  return (Math.max(1, integer(length.bars)) * beatsInMeter(meter) * 60) / clampBpm(bpm)
}

function clampLength(length: Length): Length {
  if (!length || length.unit === "bars") {
    const bars = length && length.unit === "bars" ? length.bars : 1
    return { unit: "bars", bars: Math.max(1, integer(bars)) }
  }
  return { unit: "time", seconds: Math.max(1, integer(length.seconds)) }
}

function clampSimple(pattern: SimplePattern | undefined): SimplePattern {
  const base = pattern ?? freshSimple()
  return {
    practices: Math.max(1, integer(base.practices)),
    length: clampLength(base.length),
    restAfterSec: Math.max(0, integer(base.restAfterSec)),
    stepTempo: Boolean(base.stepTempo),
    times: Math.max(1, integer(base.times)),
    repeatRound: Boolean(base.repeatRound),
    rounds: Math.max(1, integer(base.rounds)),
    restBetweenSec: Math.max(0, integer(base.restBetweenSec)),
  }
}

function clampCycle(cycle: Cycle): Cycle {
  const next: Cycle = {
    id: cycle.id || crypto.randomUUID(),
    length: clampLength(cycle.length),
    restSec: Math.max(0, integer(cycle.restSec)),
  }
  if (cycle.bpm !== undefined) next.bpm = clampBpm(cycle.bpm)
  return next
}

export function clampSession(session: Session): Session {
  const fallback = createDefaultSession()
  const cycles = (session.cycles?.length ? session.cycles : fallback.cycles).map(clampCycle)
  return {
    name: session.name ?? "",
    color: COLORS.includes(session.color) ? session.color : "red",
    meter: METERS.includes(session.meter) ? session.meter : "4/4",
    bpm: clampBpm(session.bpm),
    bpmIncrease: Math.max(0, integer(session.bpmIncrease)),
    every: Math.max(1, integer(session.every)),
    click: CLICKS.includes(session.click) ? session.click : "click-1",
    type: session.type === "complex" ? "complex" : "simple",
    simple: clampSimple(session.simple),
    cycles,
  }
}

export function cyclesFromSimple(session: Session, nextId: () => string): Cycle[] {
  const pattern = clampSimple(session.simple)
  const steps = pattern.stepTempo ? pattern.times : 1
  const rounds = pattern.repeatRound ? pattern.rounds : 1
  const cycles: Cycle[] = []
  for (let round = 0; round < rounds; round += 1) {
    for (let step = 0; step < steps; step += 1) {
      for (let practice = 0; practice < pattern.practices; practice += 1) {
        const lastOfRound = step === steps - 1 && practice === pattern.practices - 1
        const lastOfSession = round === rounds - 1 && lastOfRound
        cycles.push({
          id: nextId(),
          length: pattern.length,
          restSec: lastOfSession ? 0 : lastOfRound ? pattern.restBetweenSec : pattern.restAfterSec,
        })
      }
    }
  }
  return cycles
}

type LegacySet = {
  durationSec?: number
  bpm?: number
  restSec?: number
  repeats?: number
  color?: SetColor
  meter?: Meter
}

type LegacySession = {
  name?: string
  meter?: Meter
  bpmIncrease?: number
  click?: ClickId
  sets?: LegacySet[]
}

export function migrateLegacy(raw: unknown, nextId: () => string = () => crypto.randomUUID()): Session | null {
  if (!raw || typeof raw !== "object" || !Array.isArray((raw as LegacySession).sets)) return null
  const legacy = raw as LegacySession
  const sets = legacy.sets ?? []
  if (sets.length === 0) return null
  const increase = Math.max(0, integer(legacy.bpmIncrease ?? 0))
  const plays = sets.flatMap((set) => {
    const repeats = Math.max(1, integer(set.repeats ?? 1))
    return Array.from({ length: repeats }, (_, index) => ({
      seconds: Math.max(1, integer(set.durationSec ?? 1)),
      bpm: Math.min(300, clampBpm(set.bpm ?? 80) + index * increase),
      restSec: Math.max(0, integer(set.restSec ?? 0)),
      color: set.color,
    }))
  })
  plays[plays.length - 1].restSec = 0
  const firstBpm = plays[0].bpm
  return clampSession({
    name: legacy.name ?? "",
    color: COLORS.includes(sets[0].color as SetColor) ? (sets[0].color as SetColor) : "red",
    meter: METERS.includes(legacy.meter as Meter) ? (legacy.meter as Meter) : "4/4",
    bpm: firstBpm,
    bpmIncrease: 0,
    every: 1,
    click: CLICKS.includes(legacy.click as ClickId) ? (legacy.click as ClickId) : "click-1",
    type: "complex",
    simple: freshSimple(),
    cycles: plays.map((play) => ({
      id: nextId(),
      length: { unit: "time", seconds: play.seconds },
      restSec: play.restSec,
      ...(play.bpm === firstBpm ? {} : { bpm: play.bpm }),
    })),
  })
}

export function isClickId(value: string): value is ClickId {
  return (CLICKS as readonly string[]).includes(value)
}
```

```ts
import { ladderBpm, lengthSeconds, sessionTitle } from "./session"
import type { Meter, Session, SetColor } from "./types"

export type SegmentKind = "count-in" | "play" | "rest"

export type Segment = {
  kind: SegmentKind
  setId: string
  label: string
  repetition: number
  bpm: number
  meter: Meter
  durationSec: number
  color: SetColor | "slate"
  setIndex: number
  cycleIndex: number
  restRole?: "after" | "between"
}

type Resolved = {
  id: string
  bpm: number
  durationSec: number
  restSec: number
  restRole?: "after" | "between"
}

function resolved(session: Session): Resolved[] {
  if (session.type === "complex") {
    return session.cycles.map((cycle, index) => {
      const bpm = cycle.bpm ?? ladderBpm(session.bpm, session.bpmIncrease, session.every, index)
      return { id: cycle.id, bpm, durationSec: lengthSeconds(cycle.length, session.meter, bpm), restSec: cycle.restSec }
    })
  }
  const steps = session.simple.stepTempo ? session.simple.times : 1
  const rounds = session.simple.repeatRound ? session.simple.rounds : 1
  const items: Resolved[] = []
  for (let round = 0; round < rounds; round += 1) {
    for (let step = 0; step < steps; step += 1) {
      for (let practice = 0; practice < session.simple.practices; practice += 1) {
        const index = items.length
        const bpm = ladderBpm(session.bpm, session.bpmIncrease, session.simple.practices, index)
        const lastOfRound = step === steps - 1 && practice === session.simple.practices - 1
        const lastOfSession = round === rounds - 1 && lastOfRound
        items.push({
          id: `simple-${index}`,
          bpm,
          durationSec: lengthSeconds(session.simple.length, session.meter, bpm),
          restSec: lastOfSession ? 0 : lastOfRound ? session.simple.restBetweenSec : session.simple.restAfterSec,
          restRole: lastOfSession ? undefined : lastOfRound ? "between" : "after",
        })
      }
    }
  }
  return items
}

export function expandSession(session: Session): Segment[] {
  const label = sessionTitle(session)
  const plays = resolved(session)
  return plays.flatMap((play, index) => {
    const segments: Segment[] = [
      {
        kind: "count-in",
        setId: play.id,
        label,
        repetition: index + 1,
        bpm: play.bpm,
        meter: "4/4",
        durationSec: (4 * 60) / play.bpm,
        color: session.color,
        setIndex: index,
        cycleIndex: index,
      },
      {
        kind: "play",
        setId: play.id,
        label,
        repetition: index + 1,
        bpm: play.bpm,
        meter: session.meter,
        durationSec: play.durationSec,
        color: session.color,
        setIndex: index,
        cycleIndex: index,
      },
    ]
    if (index < plays.length - 1 && play.restSec > 0) {
      segments.push({
        kind: "rest",
        setId: play.id,
        label: "Rest",
        repetition: index + 1,
        bpm: play.bpm,
        meter: session.meter,
        durationSec: play.restSec,
        color: "slate",
        setIndex: index,
        cycleIndex: index,
        restRole: play.restRole,
      })
    }
    return segments
  })
}
```

```ts
import { clampSession, createDefaultSession, migrateLegacy } from "./session"
import type { Session } from "./types"

export const SESSION_KEY = "mentronome.session"

export function loadSession(storage: Storage): Session {
  const raw = storage.getItem(SESSION_KEY)
  if (!raw) return createDefaultSession()
  try {
    const parsed = JSON.parse(raw) as Session & { sets?: unknown }
    if (!parsed || typeof parsed !== "object") return createDefaultSession()
    if (parsed.type === "simple" || parsed.type === "complex") return clampSession(parsed)
    return migrateLegacy(parsed) ?? createDefaultSession()
  } catch {
    return createDefaultSession()
  }
}

export function saveSession(session: Session, storage: Storage): void {
  storage.setItem(SESSION_KEY, JSON.stringify(clampSession(session)))
}
```
